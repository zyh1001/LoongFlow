def wavefunction(z1, z2, z3, z4, z5, z6,
                 alpha=1.3818, beta=0.01231, beta2=0.0, gamma=0.2577,
                 lam=0.4903, eta=10.071, delta=-0.8372):

    # (island_0 identity)  x  non-holomorphic LL-mixing screen (island_1

    d14 = z1 - z4

    d24 = z2 - z4
    d25 = z2 - z5
    d26 = z2 - z6
